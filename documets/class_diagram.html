แผนภาพแสดงโครงสร้างคลาส อัตริบิวต์ และเมธอดของระบบ (System Class Architecture)

Class Diagram (Frontend Components & Backend Controller/Model)

          classDiagram
            direction LR

            class RegisterComponent {
              +registerForm: FormGroup
              +showPassword: boolean
              +showConfirmPassword: boolean
              +isSubmitted: boolean
              +passwordMismatch: boolean
              +ngOnInit() void
              +togglePassword() void
              +toggleConfirmPassword() void
              +isFieldInvalid(fieldName: string) boolean
              +onSubmit() void
            }

            class LoginComponent {
              +loginForm: FormGroup
              +showPassword: boolean
              +isSubmitted: boolean
              +ngOnInit() void
              +togglePassword() void
              +isFieldInvalid(fieldName: string) boolean
              +onSubmit() void
            }

            class SqlCleanDirective {
              -regex: RegExp
              +onInput(event: InputEvent) void
            }

            class AuthService {
              -apiUrl: string
              +login(credentials: Object) Observable
              +register(userData: Object) Observable
            }

            class User {
              +ID: uint
              +Username: string
              +PasswordHash: string
              +CreatedAt: time.Time
            }

            class AuthController {
              +Register(c: *gin.Context) void
              +Login(c: *gin.Context) void
              +GetProfile(c: *gin.Context) void
              +GetCurrentUserID(c: *gin.Context) *uint
            }

            class AuthMiddleware {
              +GenerateToken(userID: uint, username: string) string
              +AuthMiddleware() gin.HandlerFunc
            }

            RegisterComponent ..> SqlCleanDirective : Applies
            LoginComponent ..> SqlCleanDirective : Applies
            RegisterComponent --> AuthService : Calls
            LoginComponent --> AuthService : Calls

            AuthService ..> AuthController : HTTP Request
            AuthController ..> User : Query / Create
            AuthController ..> AuthMiddleware : Generates JWT
